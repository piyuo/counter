// TOC:
//  - DeliverySummaryDisplay: read-only status view for the most recent DeliveryWorker run
//
// Display rules:
//  - No delivery yet         → single "No delivery yet" row; nothing else shown.
//  - Last delivery succeeded → time of delivery with a green icon; no error rows.
//  - Last delivery failed    → red row with time, error code, detail, client-error badge,
//                              and either a next-retry time or a "Discarded" badge.

import 'package:flutter/material.dart';
import 'package:intl/intl.dart';

class DeliverySummaryDisplay extends StatelessWidget {
  const DeliverySummaryDisplay({
    this.lastDeliveryTimeUtc,
    this.isLastDeliverySuccess,
    this.lastError,
    this.lastErrorDetail,
    this.lastClientError,
    this.nextRetryTimeUtc,
    super.key,
  });

  /// UTC timestamp of the most recent delivery attempt (success or failure).
  final DateTime? lastDeliveryTimeUtc;

  /// Outcome of the most recent attempt; null means no attempt has happened yet.
  final bool? isLastDeliverySuccess;

  /// Machine-readable error code (e.g. `transport_exception`).
  final String? lastError;

  /// Human-readable error detail.
  final String? lastErrorDetail;

  /// `true` = 4xx client error, `false` = network / server error.
  final bool? lastClientError;

  /// Earliest scheduled retry; null on success or when all items were discarded.
  final DateTime? nextRetryTimeUtc;

  @override
  Widget build(BuildContext context) {
    // ── No delivery yet ────────────────────────────────────────────────────
    if (lastDeliveryTimeUtc == null) {
      return _StatusRow(icon: Icons.cloud_off_outlined, color: Colors.grey, label: 'No delivery yet');
    }

    final timeLabel = _formatTime(lastDeliveryTimeUtc!.toLocal());

    // ── Success ────────────────────────────────────────────────────────────
    if (isLastDeliverySuccess == true) {
      return _StatusRow(icon: Icons.cloud_done_outlined, color: Colors.green, label: 'Delivered at $timeLabel');
    }

    // ── Failure ────────────────────────────────────────────────────────────
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      mainAxisSize: MainAxisSize.min,
      children: [
        _StatusRow(icon: Icons.cloud_off_outlined, color: Colors.red, label: 'Failed at $timeLabel'),
        if (lastError != null) _DetailRow(icon: Icons.error_outline, text: lastError!),
        if (lastErrorDetail != null) _DetailRow(icon: Icons.info_outline, text: lastErrorDetail!),
        if (lastClientError == true) _DetailRow(icon: Icons.report_outlined, text: 'Client error (4xx)'),
        if (nextRetryTimeUtc != null)
          _DetailRow(icon: Icons.schedule, text: 'Retry at ${_formatTime(nextRetryTimeUtc!.toLocal())}')
        else
          const _DetailRow(icon: Icons.block, text: 'Permanently discarded'),
      ],
    );
  }

  static String _formatTime(DateTime local) {
    final now = DateTime.now();
    final isToday = local.year == now.year && local.month == now.month && local.day == now.day;
    return isToday ? DateFormat.jm().format(local) : DateFormat.MMMd().add_jm().format(local);
  }
}

// ---------------------------------------------------------------------------
// Private helpers
// ---------------------------------------------------------------------------

class _StatusRow extends StatelessWidget {
  const _StatusRow({required this.icon, required this.color, required this.label});

  final IconData icon;
  final Color color;
  final String label;

  @override
  Widget build(BuildContext context) {
    return Row(
      children: [
        Icon(icon, color: color, size: 18),
        const SizedBox(width: 8),
        Flexible(
          child: Text(label, style: TextStyle(color: color)),
        ),
      ],
    );
  }
}

class _DetailRow extends StatelessWidget {
  const _DetailRow({required this.icon, required this.text});

  final IconData icon;
  final String text;

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.only(left: 26, top: 4),
      child: Row(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Icon(icon, size: 14, color: Colors.grey),
          const SizedBox(width: 6),
          Flexible(
            child: Text(text, style: const TextStyle(fontSize: 12, color: Colors.grey)),
          ),
        ],
      ),
    );
  }
}
